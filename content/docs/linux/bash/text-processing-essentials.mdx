Text processing is where bash really shines. Log files, configuration data, CSV exports, and API responses all need parsing and transformation. Learning `sed`, `awk`, and `grep` lets you extract and manipulate data without spawning additional processes or reaching for Python.

## grep: Filtering and Pattern Matching

`grep` is your primary tool for finding lines matching a pattern. Master the common flags:

```bash
#!/bin/bash
set -euo pipefail

# Basic grep: Find lines containing a pattern
grep "ERROR" application.log

# -i: Case-insensitive search
grep -i "error" application.log

# -v: Invert match (lines NOT containing pattern)
grep -v "DEBUG" application.log | grep "ERROR"

# -E: Extended regex (like egrep)
grep -E "ERROR|FATAL|CRITICAL" application.log

# -n: Show line numbers
grep -n "database connection failed" application.log

# -c: Count matching lines
error_count=$(grep -c "ERROR" application.log)
echo "Total errors: $error_count"

# -l: Show only filenames that match
find /var/log -name "*.log" -type f | xargs grep -l "PANIC"

# Practical: Extract errors from logs with context
extract_errors() {
  local log_file="$1"
  local context_lines="${2:-3}"  # Show 3 lines of context

  echo "=== Errors in $log_file ==="
  grep -B "$context_lines" -A "$context_lines" "ERROR" "$log_file" | head -100
}

extract_errors "app.log" 2
```

## sed: Stream Editing

`sed` modifies text in a stream. Use it for substitutions, deletions, and transformations:

```bash
#!/bin/bash
set -euo pipefail

# Basic substitution: Replace first occurrence on each line
echo "database_url=http://localhost database_url=bad" | sed 's/database_url=.*/database_url=postgresql:\/\/prod/'
# Output: database_url=postgresql://prod

# Global replacement: All occurrences on a line
echo "old old old" | sed 's/old/new/g'
# Output: new new new

# Case-insensitive replacement
echo "DATABASE url" | sed 's/database/db/i'
# Output: db url

# Delete lines matching a pattern
sed '/^#/d' config.ini  # Remove comments
sed '/^$/d' config.ini  # Remove blank lines

# Extract specific range of lines
sed -n '10,20p' large_file.txt

# In-place editing (creates backup first)
sed -i.bak 's/old_hostname/new_hostname/g' /etc/hosts

# Practical: Extract configuration from a file
extract_config_section() {
  local config_file="$1"
  local section="$2"

  # Find section and extract until next section or EOF
  sed -n "/^\[$section\]/,/^\[/p" "$config_file" | sed '$d'  # Remove last empty line
}

# Practical: Escape special characters for sed replacements
escape_for_sed() {
  local text="$1"
  # Escape &, \, and /
  echo "$text" | sed 's/[&\/]/\\&/g'
}

# Use escaped text in substitution
new_url="https://api.example.com/path"
escaped=$(escape_for_sed "$new_url")
sed -i "s|^api_url=.*|api_url=$escaped|" config.ini
```

## awk: Column and Data Processing

`awk` is powerful for structured data (CSV, logs, config files with consistent format). It breaks each line into fields:

```bash
#!/bin/bash
set -euo pipefail

# Basic: Print specific column
# Access fields with $1, $2, $3, etc. ($0 is the whole line)
ps aux | awk '{print $1, $2, $11}'  # Print user, PID, command
# Output: root 1 /sbin/init

# CSV processing
awk -F',' '{print $2}' users.csv  # Print second column (-F sets delimiter)

# Conditional processing
awk '$3 > 100 {print $1, $3}' sales.txt  # Print lines where 3rd column > 100

# BEGIN and END blocks (run before/after processing)
awk '
  BEGIN { total = 0 }
  { total += $3 }  # Add 3rd column to total
  END { print "Total:", total }
' numbers.txt

# String functions
awk '{print toupper($1)}' names.txt  # Convert first field to uppercase
awk '{print length($0)}' file.txt    # Print line length
awk '{print substr($1, 1, 5)}' file.txt  # First 5 characters

# Practical: Parse JSON-like logs
parse_access_log() {
  local log_file="$1"

  # Log format: [timestamp] client_ip method path response_code bytes response_time
  awk '
    $NF > 1000 {  # If response time > 1000ms
      print "[SLOW] " $1 " " $2 " " $3 " " $4 " took " $NF "ms"
    }
    $5 >= 400 {   # If response code >= 400
      print "[ERROR] " $2 " returned " $5
    }
  ' "$log_file"
}

# Practical: Sum resource usage
sum_memory_usage() {
  ps aux | awk '
    NR > 1 {  # Skip header
      total_mem += $6
      count++
    }
    END {
      print "Total memory: " total_mem / 1024 " MB across " count " processes"
    }
  '
}

parse_access_log "access.log"
sum_memory_usage
```

## Complex Data Extraction

Combine tools for powerful data transformations:

```bash
#!/bin/bash
set -euo pipefail

# Extract URLs from log file, filter duplicates, count frequency
extract_popular_endpoints() {
  local log_file="$1"

  grep "GET" "$log_file" \
    | awk '{print $6}' \
    | sort \
    | uniq -c \
    | sort -rn \
    | head -20

  # Output: 1523 /api/users, 987 /api/posts, ...
}

# Parse JSON logs (simple key-value extraction)
extract_json_field() {
  local json_input="$1"
  local field_name="$2"

  echo "$json_input" | grep -o "\"$field_name\":\"[^\"]*\"" | cut -d'"' -f4
}

# Practical: Extract request traces from logs
get_request_timeline() {
  local request_id="$1"
  local log_file="${2:-application.log}"

  grep "request_id=$request_id" "$log_file" \
    | awk -F'[|]' '{print $1, $NF}' \
    | sed 's/\[//' \
    | sort -u
}

extract_popular_endpoints "access.log"
echo '{"user":"alice","id":123}' | extract_json_field '{"user":"alice","id":123}' "user"
get_request_timeline "req-12345"
```

## Anti-patterns and Performance

Know the limitations of text processing tools:

```bash
#!/bin/bash
set -euo pipefail

# Anti-pattern: Over-processing
bad_example() {
  local file="$1"

  # Anti-pattern: Multiple passes through the file
  total_lines=$(wc -l < "$file")
  error_count=$(grep -c "ERROR" "$file")
  warning_count=$(grep -c "WARNING" "$file")
  # This scans the file 3 times
}

# Better: Single pass
good_example() {
  local file="$1"

  awk '
    /ERROR/ { errors++ }
    /WARNING/ { warnings++ }
    { total++ }
    END {
      print "Total lines:", total
      print "Errors:", errors + 0
      print "Warnings:", warnings + 0
    }
  ' "$file"
}

# Anti-pattern: Piping grep through grep
bad_filter() {
  grep "ERROR" app.log | grep "database"
}

# Better: Single grep with regex
good_filter() {
  grep -E "ERROR.*database" app.log
}

# When to use bash vs Python
# Bash is efficient for simple transformations
# Python is better for complex logic

# Simple: Extract and count (use bash)
grep -o "status=[0-9]*" logs/* | cut -d= -f2 | sort | uniq -c

# Complex: Parse, transform, logic (consider Python)
# - Parsing structured formats (XML, complex JSON)
# - Complex conditional logic
# - Need for external libraries
```

## Production Best Practices

**1. Quote variable expansions** — `grep "$pattern"` not `grep $pattern`

**2. Use `-F` delimiter in awk** — Explicit about field separator

**3. Check tool availability** — `awk`, `sed`, and `grep` have variants (`mawk`, `gawk`, `busybox`)

**4. Test patterns carefully** — Off-by-one errors with field indices are common

**5. Prefer single-pass processing** — Pipe data once, not multiple times

**6. Consider line endings** — Windows files (CRLF) behave differently

**7. Know when to use Python** — Complex logic, structured formats (XML, JSON), libraries

**8. Document regex patterns** — Bash regex syntax can be cryptic

These tools handle text efficiently. Learn them well, and you'll write powerful data transformation scripts without external dependencies.
