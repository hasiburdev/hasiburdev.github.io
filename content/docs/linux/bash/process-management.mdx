Modern bash scripts often manage long-running processes: services, workers, data pipelines. Understanding process signals, job control, and graceful shutdown is essential for reliable automation, especially in containerized environments where processes receive SIGTERM and must clean up quickly.

## Process IDs and Job Control

Bash tracks foreground and background processes. Learn to work with them:

```bash
#!/bin/bash
set -euo pipefail

# Start process in background
long_running_command &
bg_process_pid=$!  # Capture PID of last background job

echo "Background process PID: $bg_process_pid"

# Wait for background process
wait "$bg_process_pid"
exit_code=$?

if [[ $exit_code -eq 0 ]]; then
  echo "Background process completed successfully"
else
  echo "Background process failed with exit code: $exit_code" >&2
fi

# Run multiple processes in parallel
run_parallel_tasks() {
  local pids=()

  # Start tasks
  task_one &
  pids+=($!)

  task_two &
  pids+=($!)

  task_three &
  pids+=($!)

  # Wait for all to complete
  local failed=0
  for pid in "${pids[@]}"; do
    if ! wait "$pid"; then
      echo "Task PID $pid failed" >&2
      failed=$((failed + 1))
    fi
  done

  if [[ $failed -gt 0 ]]; then
    echo "ERROR: $failed tasks failed" >&2
    return 1
  fi
}

# Check if process is still running
is_process_running() {
  local pid="$1"
  kill -0 "$pid" 2>/dev/null  # Signal 0 checks without sending
}

run_parallel_tasks
```

## Signals and Graceful Shutdown

Processes receive signals from the OS. The most important are SIGTERM (request to terminate) and SIGINT (interrupt, usually Ctrl+C). Handle them gracefully:

```bash
#!/bin/bash
set -euo pipefail

# Global state for cleanup
background_processes=()
temp_files=()

# Cleanup handler - called on signal or exit
cleanup() {
  local exit_code=$?
  echo "Shutting down gracefully (exit code: $exit_code)..."

  # Kill background processes gracefully
  for pid in "${background_processes[@]}"; do
    if kill -0 "$pid" 2>/dev/null; then
      echo "Terminating process $pid..."
      kill -TERM "$pid"

      # Wait up to 5 seconds for graceful shutdown
      local count=0
      while kill -0 "$pid" 2>/dev/null && [[ $count -lt 5 ]]; do
        sleep 1
        count=$((count + 1))
      done

      # Force kill if still running
      if kill -0 "$pid" 2>/dev/null; then
        echo "Force killing process $pid..."
        kill -9 "$pid"
      fi
    fi
  done

  # Clean up temp files
  for tmpfile in "${temp_files[@]}"; do
    if [[ -f "$tmpfile" ]]; then
      rm -f "$tmpfile"
    fi
  done

  return "$exit_code"
}

# Register cleanup handler
trap cleanup EXIT

# Handle SIGTERM from container orchestrator
# (Docker/Kubernetes sends SIGTERM before shutdown)
handle_sigterm() {
  echo "Received SIGTERM - initiating graceful shutdown..."
  exit 0  # Triggers cleanup trap
}

trap handle_sigterm SIGTERM

# Example: Long-running service
run_service() {
  local service_name="$1"
  echo "Starting service: $service_name"

  # Create temp state file
  local state_file
  state_file=$(mktemp)
  temp_files+=("$state_file")

  # Start background worker
  long_running_worker "$service_name" &
  local worker_pid=$!
  background_processes+=("$worker_pid")

  echo "Worker PID: $worker_pid"

  # Wait for worker
  wait "$worker_pid"
}

long_running_worker() {
  local service="$1"
  local count=0

  while [[ $count -lt 100 ]]; do
    echo "[$service] Working... ($count)"
    sleep 5
    count=$((count + 1))
  done
}

run_service "my-app"
```

## Managing Child Processes

When your script starts processes, it's responsible for managing them:

```bash
#!/bin/bash
set -euo pipefail

# Anti-pattern: Orphan processes
start_services_bad() {
  nginx &          # PID saved nowhere, can't stop it
  postgres &
  redis &
}

# Better: Track and manage
start_services_good() {
  local -A services

  echo "Starting services..."

  nginx &
  services[nginx]=$!

  postgres &
  services[postgres]=$!

  redis &
  services[redis]=$!

  # Monitor services
  while true; do
    for name in "${!services[@]}"; do
      local pid=${services[$name]}
      if ! kill -0 "$pid" 2>/dev/null; then
        echo "WARNING: Service $name (PID $pid) has stopped"
        # Restart or fail?
      fi
    done
    sleep 10
  done
}

# Process groups: Kill all child processes at once
run_with_process_group() {
  # Create new process group
  local script="${1:-./worker.sh}"

  # Start with process group
  setsid "$script" &
  local group_pid=$!

  # Kill entire process group when needed
  kill_group() {
    echo "Killing process group $group_pid..."
    kill -TERM -- "-$group_pid"  # Negative PID kills whole group
  }

  trap kill_group EXIT
  wait "$group_pid"
}
```

## Process Substitution

Process substitution (`<(...)`) lets you treat command output as a file, or feed input from multiple sources:

```bash
#!/bin/bash
set -euo pipefail

# Compare output of two commands
diff <(command_one) <(command_two)

# Practical: Compare config files
compare_configs() {
  local dev_env="/etc/myapp/dev.env"
  local prod_env="/etc/myapp/prod.env"

  # Show differences
  diff <(sort "$dev_env") <(sort "$prod_env") || true
}

# Feed multiple file contents as arguments
combine_logs() {
  local log_files=(/var/log/app/*.log)

  # Process all logs as single input
  sort <(cat "${log_files[@]}") | tail -100
}

# Parallel processing
analyze_multiple_services() {
  local services=("auth" "api" "worker")

  # Run checks in parallel and capture output
  paste \
    <(for svc in "${services[@]}"; do echo "Checking $svc"; done) \
    <(for svc in "${services[@]}"; do systemctl status "$svc"; done)
}

compare_configs
```

## Timeout and Long-Running Processes

Prevent scripts from hanging indefinitely:

```bash
#!/bin/bash
set -euo pipefail

# Use timeout command
run_with_timeout() {
  local timeout_seconds="$1"
  local command="${@:2}"

  # timeout will kill command if it exceeds time limit
  timeout "$timeout_seconds" "$command" || {
    local exit_code=$?
    if [[ $exit_code -eq 124 ]]; then
      echo "ERROR: Command timed out after $timeout_seconds seconds" >&2
      return 1
    fi
    return "$exit_code"
  }
}

# Manual timeout with sleep in background
wait_with_timeout() {
  local timeout_seconds="$1"
  local pid="$2"

  # Start sleep that will kill the process if timeout expires
  (sleep "$timeout_seconds" && kill -9 "$pid" 2>/dev/null) &
  local sleep_pid=$!

  # Wait for main process
  if wait "$pid"; then
    # Process succeeded, kill the sleep
    kill "$sleep_pid" 2>/dev/null || true
    return 0
  else
    local exit_code=$?
    kill "$sleep_pid" 2>/dev/null || true
    return "$exit_code"
  fi
}

# Practical: Health check with timeout
health_check() {
  local service_url="$1"
  local timeout="${2:-5}"

  if curl -s --max-time "$timeout" "$service_url" >/dev/null 2>&1; then
    echo "Service is healthy"
    return 0
  else
    echo "Service health check failed" >&2
    return 1
  fi
}

run_with_timeout 30 sleep 10
health_check "http://localhost:8080/health" 10
```

## Container and Orchestration Awareness

Modern scripts run in containers. Understand signals and PID 1:

```bash
#!/bin/bash
set -euo pipefail

# Container init script - acts as PID 1
# Kubernetes/Docker sends SIGTERM before container shutdown

handle_shutdown() {
  echo "Container shutdown signal received"

  # Stop the application
  echo "Stopping application..."
  kill -TERM "$app_pid" 2>/dev/null || true

  # Wait for graceful shutdown
  local count=0
  while kill -0 "$app_pid" 2>/dev/null && [[ $count -lt 30 ]]; do
    sleep 1
    count=$((count + 1))
  done

  # Force kill if still running
  kill -9 "$app_pid" 2>/dev/null || true

  exit 0
}

trap handle_shutdown SIGTERM

# Start application
./myapp &
app_pid=$!

echo "Application started with PID $app_pid"
wait "$app_pid"
```

## Production Best Practices

**1. Always track background process PIDs** — You can't manage what you don't track.

**2. Use `trap` for signal handling** — Ensure cleanup on exit, regardless of cause.

**3. Implement graceful shutdown** — Allow processes time to finish work before forcing shutdown.

**4. Monitor background processes** — Check periodically if they're still running.

**5. Use `timeout` for potentially hanging commands** — Prevent scripts from blocking indefinitely.

**6. Handle SIGTERM in containers** — Container orchestrators expect graceful shutdown.

**7. Clean up resources** — Kill processes, remove temp files, close connections in cleanup handlers.

**8. Test failure scenarios** — What happens if a background process dies? Script restarts? Fails?

Process management is often overlooked but critical for production reliability. Invest in handling it properly.
